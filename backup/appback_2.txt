def parse_schedule_range(utterance: str, today: date):
    u = (utterance or "").strip().replace(" ", "").lower()

    if "오늘학사일정" in u or u == "오늘":
        return today, today, "오늘 학사일정"
    
    if "내일학사일정" in u or u == "내일":
        tomorrow = today + timedelta(days=1)
        return tomorrow, tomorrow, "내일 학사일정"
    
    if "이번달학사일정" in u or "이번달" in u:
        start = date(today.year, today.month, 1)
        last_day = calendar.monthrange(today.year, today.month)[1]
        end = date(today.year, today.month, last_day)
        return start, end, f"{today.year}년 {today.month}월 학사일정"

    if "다음달학사일정" in u or "다음달" in u:
        year = today.year + (1 if today.month == 12 else 0)
        month = 1 if today.month == 12 else today.month + 1
        start = date(year, month, 1)
        last_day = calendar.monthrange(year, month)[1]
        end = date(year, month, last_day)
        return start, end, f"{year}년 {month}월 학사일정"

    m = re.search(r'(?:(20\d{2})년)?(\d{1,2})월학사일정', u)
    if m:
        y = int(m.group(1)) if m.group(1) else today.year
        mon = int(m.group(2))
        start = date(y, mon, 1)
        last_day = calendar.monthrange(y, mon)[1]
        end = date(y, mon, last_day)
        return start, end, f"{y}년 {mon}월 학사일정"

    if "학사일정" in u:
        return None, None, "학사일정 전체"

    return None, None, ""


def fetch_academic_events(conn, start: date|None, end: date|None):
    """
    기간 교집합 조건:
      start_date <= (요청 end) AND end_date >= (요청 start)
    start/end가 None이면 전체
    """
    cursor = conn.cursor()
    if start and end:
        sql = """
        SELECT title, start_date, end_date
        FROM dbo.academic_event
        WHERE start_date <= ? AND end_date >= ?
        ORDER BY start_date, title
        """
        cursor.execute(sql, end, start)
    else:
        sql = """
        SELECT title, start_date, end_date
        FROM dbo.academic_event
        ORDER BY start_date, title
        """
        cursor.execute(sql)
    rows = cursor.fetchall()
    cursor.close()
    return rows


def format_event_line(title: str, sd, ed) -> str:
    # sd, ed: datetime.date 또는 datetime
    if isinstance(sd, datetime):
        sd = sd.date()
    if isinstance(ed, datetime):
        ed = ed.date()
    if sd == ed:
        return f"{sd:%Y-%m-%d} · {title}"
    else:
        return f"{sd:%Y-%m-%d} ~ {ed:%Y-%m-%d} · {title}"


def chunk_lines(lines, chunk_size=20):
    for i in range(0, len(lines), chunk_size):
        yield lines[i:i+chunk_size]


def make_text_outputs_from_lines(lines, title_label: str, page: int, page_size: int):
    """
    lines 전체를 page/page_size로 끊어 simpleText outputs + quickReplies(다음/이전) 구성
    """
    total = len(lines)
    total_pages = max(1, ceil(total / page_size))
    page = max(1, min(page, total_pages))
    start_idx = (page - 1) * page_size
    end_idx = min(total, start_idx + page_size)
    slice_lines = lines[start_idx:end_idx]

    outputs = []
    header = f"📅 {title_label}"
    body = header + "\n" + "\n".join(slice_lines)
    outputs.append({"simpleText": {"text": body}})

    qrs = []
    if page > 1:
        qrs.append({
            "action": "message",
            "label": "◀ 이전",
            "messageText": f"{title_label} p:{page-1}"
        })
    if page < total_pages:
        qrs.append({
            "action": "message",
            "label": "다음 ▶",
            "messageText": f"{title_label} p:{page+1}"
        })
    return outputs, qrs
